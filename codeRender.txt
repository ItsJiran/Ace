                  {activeTab === 'response' && (
                        <div className="absolute inset-0 overflow-y-auto overflow-x-hidden p-3 space-y-3">
                            {!responseMemory?.response_turns || responseMemory.response_turns.length === 0 ? (
                                <div className="text-[11px] text-zinc-500 italic p-4 text-center border cursor-not-allowed border-zinc-800/50 rounded bg-black/20">
                                    No response memory turns recorded...
                                </div>
                            ) : (
                                <div className="flex flex-col gap-2">
                                    {responseMemory.response_turns.map((t, idx) => (
                                        <TurnAccordion
                                            key={t.turn_id}
                                            turn={t}
                                            turnIndex={idx + 1}
                                            isActive={activeResponseTurnId === t.turn_id}
                                            onToggle={() => {
                                                if (activeResponseTurnId === t.turn_id) {
                                                    setActiveResponseTurnId(null);
                                                } else {
                                                    setActiveResponseTurnId(t.turn_id);
                                                    const acts = t.attempts || [];
                                                    if (acts.length > 0) {
                                                        const latestIdx = acts.reduce((max, a) => Math.max(max, a.attempt_index), 0);
                                                        setActiveResponseAttempt(latestIdx);
                                                    } else {
                                                        setActiveResponseAttempt(1);
                                                    }
                                                }
                                            }}
                                            session={session}
                                            responseMemory={responseMemory}
                                        />
                                    ))}
                                </div>
                            )}
                        </div>
                    )}
